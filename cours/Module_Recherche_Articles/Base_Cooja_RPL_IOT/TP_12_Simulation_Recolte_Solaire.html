TP 12 — Simulation avec Récolte d'Énergie Solaire

TP 12 — Durée estimée : 2h30 — Prérequis : Cours 07, TP 07

1. Objectifs

  • Implémenter un modèle de récolte d'énergie solaire dans un nœud Contiki-NG
  • Configurer le duty-cycling pour économiser l'énergie
  • Comparer la durée de vie des nœuds avec et sans récolte d'énergie
  • Utiliser PowerTracker pour vérifier la consommation

2. Modèle de batterie

Avant d'ajouter la récolte, définissons un modèle de batterie simple :

#define BATTERY_CAPACITY 2000  // mAh
#define VOLTAGE 3.0            // Volts
#define INITIAL_CHARGE 0.9     // 90% initial

// Énergie totale disponible (en mJ)
// 1 mAh × 3.6 = 3.6 C (Coulombs)
// E = V × I × t = 3.0 × 0.002 × 3600 = 21.6 J = 21600 mJ
#define TOTAL_ENERGY_mJ (BATTERY_CAPACITY * VOLTAGE * 3.6)

3. Module de récolte d'énergie solaire

Créez solar_harvest.h :

#ifndef SOLAR_HARVEST_H_
#define SOLAR_HARVEST_H_

void solar_harvest_init(void);
int solar_get_power(void);         // puissance reçue en mW
int solar_get_energy_collected(void);  // énergie collectée en mJ

#endif

Créez solar_harvest.c :

#include "solar_harvest.h"
#include "lib/random.h"
#include 

static int total_energy_collected = 0;
static int panel_power = 10;  // 10 mW (panneau solaire typique 2cm²)

// Cycle jour/nuit simulé (jour = 50s, nuit = 30s)
#define DAY_DURATION 50
#define NIGHT_DURATION 30
static int cycle_time = 0;

void solar_harvest_init(void) {
  printf("Module récolte solaire initialisé\n");
  cycle_time = 0;
  total_energy_collected = 0;
}

int solar_get_power(void) {
  cycle_time++;
  if(cycle_time <= DAY_DURATION) {
    // Jour : puissance variable selon l'ensoleillement
    int variation = (random_rand() % 40) - 20;  // +/- 20%
    int power = panel_power + (panel_power * variation / 100);
    return (power > 1) ? power : 1;
  } else {
    // Nuit : pas d'énergie solaire
    if(cycle_time >= DAY_DURATION + NIGHT_DURATION) {
      cycle_time = 0;
    }
    return 0;
  }
}

int solar_get_energy_collected(void) {
  return total_energy_collected;
}

// À appeler périodiquement
void solar_harvest_update(void) {
  int power = solar_get_power();
  // 1 tick = 1 ms, récolte = puissance × temps
  total_energy_collected += power;  // approximation simple
}

4. Intégration dans le nœud client

Ajoutez la récolte d'énergie au client UDP :

#include "solar_harvest.h"

PROCESS_THREAD(client_process, ev, data)
{
  static struct etimer harvest_timer;
  PROCESS_BEGIN();

  solar_harvest_init();

  // Timer pour la récolte (toutes les secondes)
  etimer_set(&harvest_timer, CLOCK_SECOND);

  while(1) {
    PROCESS_YIELD();
    if(ev == PROCESS_EVENT_TIMER && &harvest_timer == data) {
      solar_harvest_update();
      int power = solar_get_power();
      if(power > 0) {
        printf("RÉCOLTE: puissance=%d mW, total=%d mJ\n",
               power, solar_get_energy_collected());
      }
      etimer_restart(&harvest_timer);
    }
  }

  PROCESS_END();
}

5. Configuration du duty-cycling

Dans project-conf.h, activez ContikiMAC pour le duty-cycling :

// Duty-cycling avec ContikiMAC
#define NETSTACK_CONF_MAC contikimac_driver
#define CONTIKIMAC_CONF_ANNOUNCEMENTS 0

// Paramètres du duty-cycling
#define NETSTACK_CONF_RDC contikimac_driver
#define CONTIKIMAC_CONF_CHANNEL_CHECK_RATE 8  // vérifications/s

  • PDR et latence

7. Analyse de la durée de vie

Calculez la durée de vie estimée :

# Durée de vie sans récolte :
Energie_totale_initiale = BATTERY_CAPACITY × VOLTAGE × 3.6
Puissance_moyenne = Total_energy / Temps_simulation
Duree_vie_sans_recolte = Energie_totale_initiale / Puissance_moyenne

# Durée de vie avec récolte :
Puissance_nette = Puissance_moyenne - Puissance_recoltee_moyenne
Duree_vie_avec_recolte = Energie_totale_initiale / max(Puissance_nette, 0.001)
