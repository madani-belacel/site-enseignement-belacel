TP 10 — Ajout de Capteurs Simulés (Température, Humidité)

TP 10 — Durée estimée : 2h — Prérequis : TP 09

1. Objectifs

  • Implémenter des capteurs simulés de température et d'humidité dans Contiki-NG
  • Utiliser des valeurs aléatoires pour simuler les lectures de capteurs
  • Envoyer les données capteurs via UDP vers le serveur
  • Afficher et collecter les données côté serveur

2. Principe des capteurs simulés

Dans Cooja, les nœuds Sky ne sont pas connectés à du matériel réel. Nous simulons donc les lectures de capteurs avec des valeurs aléatoires qui imitent le comportement de vrais capteurs :

  • Température : 15-35 °C (variation lente avec bruit)
  • Humidité : 40-80 % (variation lente)
  • Luminosité : 100-1000 lux (variation aléatoire)

3. Module capteur simulé

Créez un fichier simulated_sensor.h :

#ifndef SIMULATED_SENSOR_H_
#define SIMULATED_SENSOR_H_

typedef struct {
  int temperature;      // en dixièmes de °C (ex: 235 = 23.5°C)
  int humidity;         // en dixièmes de % (ex: 655 = 65.5%)
  int light;            // en lux
  unsigned long timestamp;
} sensor_data_t;

void sensor_init(void);
sensor_data_t sensor_read(void);

#endif /* SIMULATED_SENSOR_H_ */

Créez le fichier simulated_sensor.c :

#include "simulated_sensor.h"
#include "lib/random.h"
#include 

static int base_temp = 250;   // 25.0°C
static int base_hum  = 600;   // 60.0%
static int base_light = 500;  // 500 lux

void sensor_init(void) {
  random_init(42);  // seed fixe pour reproductibilité
  printf("Capteurs simulés initialisés\n");
}

sensor_data_t sensor_read(void) {
  sensor_data_t data;

  // Variation aléatoire autour de la valeur de base
  int temp_var = (random_rand() % 100) - 50;    // +/- 5°C
  int hum_var  = (random_rand() % 100) - 50;    // +/- 5%
  int light_var = (random_rand() % 400) - 200;  // +/- 200 lux

  data.temperature = base_temp + temp_var;
  data.humidity = base_hum + hum_var;
  data.light = base_light + light_var;
  data.timestamp = clock_time();

  // Faire varier lentement la base (simuler cycle jour/nuit)
  base_temp += (random_rand() % 10) - 5;
  base_hum  += (random_rand() % 10) - 5;
  base_light += (random_rand() % 50) - 25;

  // Contraintes réalistes
  if(base_temp < 150) base_temp = 150;
  if(base_temp > 350) base_temp = 350;
  if(base_hum < 400) base_hum = 400;
  if(base_hum > 800) base_hum = 800;
  if(base_light < 50) base_light = 50;
  if(base_light > 1000) base_light = 1000;

  return data;
}

4. Client UDP avec capteurs

Modifiez le client UDP pour intégrer les capteurs :

#include "contiki.h"
#include "net/ip/uip.h"
#include "net/ipv6/uip-ds6.h"
#include "net/ip/uip-udp-packet.h"
#include "simulated_sensor.h"
#include 
#include 

#define UDP_PORT 8765
#define SEND_INTERVAL (CLOCK_SECOND * 30)

PROCESS(sensor_client_process, "Client Capteurs");
AUTOSTART_PROCESSES(&sensor_client_process);

static struct uip_udp_conn *client_conn;
static struct etimer et;

PROCESS_THREAD(sensor_client_process, ev, data)
{
  uip_ipaddr_t server_ip;
  char buf[128];
  sensor_data_t sensor;
  PROCESS_BEGIN();

  sensor_init();
  uip_ip6addr(&server_ip, 0xfe80, 0, 0, 0, 0, 0, 0, 0);
  client_conn = udp_new(&server_ip, UIP_HTONS(UDP_PORT), NULL);
  udp_bind(client_conn, UIP_HTONS(UDP_PORT));

  printf("Client capteurs démarré (ID=%d)\n", node_id);
  etimer_set(&et, SEND_INTERVAL);

  while(1) {
    PROCESS_YIELD();
    if(ev == PROCESS_EVENT_TIMER) {
      sensor = sensor_read();

      snprintf(buf, sizeof(buf),
        "ID=%d|TEMP=%d.%d|HUM=%d.%d|LUX=%d|TIME=%lu",
        node_id,
        sensor.temperature / 10, sensor.temperature % 10,
        sensor.humidity / 10, sensor.humidity % 10,
        sensor.light,
        (unsigned long)sensor.timestamp);

      printf("ENVOI CAPTEUR: %s\n", buf);
      udp_packet_send(client_conn, buf, strlen(buf));
      etimer_restart(&et);
    }
    else if(ev == tcpip_event && uip_newdata()) {
      printf("Réponse: %s\n", (char*)uip_appdata);
    }
  }

  PROCESS_END();
}

5. Serveur de collecte de données

Le serveur affiche les données reçues de chaque capteur :

// Extrait de la boucle principale
while(1) {
  PROCESS_YIELD();
  if(ev == tcpip_event && uip_newdata()) {
    printf("[%lu] DONNÉES: %s\n",
           (unsigned long)clock_time(), (char*)uip_appdata);
    // Les données sont au format CSV, prêtes pour l'analyse
  }
}

8. Exercices

  1. Ajoutez un capteur de qualité de l'air (CO2) simulé entre 400 et 2000 ppm
  2. Implémentez un seuil d'alerte : si température > 35°C, envoyez un message d'alerte prioritaire
  3. Du côté serveur, calculez la moyenne des températures reçues toutes les 5 minutes
  4. Exportez les données collectées dans un format CSV analysable par Python
