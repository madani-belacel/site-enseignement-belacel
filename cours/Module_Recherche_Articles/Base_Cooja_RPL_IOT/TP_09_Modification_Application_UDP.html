TP 09 — Modification d'une Application UDP dans Contiki-NG

TP 09 — Durée estimée : 2h — Prérequis : Cours 06, TP 03

1. Objectifs

  • Étendre l'application UDP client/serveur de Contiki-NG
  • Ajouter un numéro de séquence dans les paquets UDP
  • Implémenter un mécanisme stop-and-wait simple
  • Ajouter des données de payload personnalisées

2. Structure du projet

Copiez l'exemple rpl-udp comme base :

cp -r $CONTIKI/examples/rpl-udp ~/mon_app_udp_avancee
cd ~/mon_app_udp_avancee

3. Client UDP amélioré avec numéros de séquence

Modifiez udp-client.c :

#include "contiki.h"
#include "net/ip/uip.h"
#include "net/ipv6/uip-ds6.h"
#include "net/ip/uip-udp-packet.h"
#include 
#include 

#define UDP_PORT 8765
#define SEND_INTERVAL (CLOCK_SECOND * 10)

PROCESS(udp_client_process, "UDP Client amélioré");
AUTOSTART_PROCESSES(&udp_client_process);

static struct uip_udp_conn *client_conn;
static struct etimer periodic_timer;
static struct etimer ack_timer;
static int seqno = 0;
static int waiting_for_ack = 0;
static int retransmissions = 0;
#define MAX_RETRANSMISSIONS 3

PROCESS_THREAD(udp_client_process, ev, data)
{
  uip_ipaddr_t server_ip;
  char buf[64];
  PROCESS_BEGIN();

  uip_ip6addr(&server_ip, 0xfe80, 0, 0, 0, 0, 0, 0, 0);
  client_conn = udp_new(&server_ip, UIP_HTONS(UDP_PORT), NULL);
  udp_bind(client_conn, UIP_HTONS(UDP_PORT));

  printf("Client UDP avancé démarré (ID=%d)\n", node_id);
  etimer_set(&periodic_timer, SEND_INTERVAL);

  while(1) {
    PROCESS_YIELD();

    if(ev == PROCESS_EVENT_TIMER) {
      if(&periodic_timer == data && !waiting_for_ack) {
        seqno++;
        snprintf(buf, sizeof(buf),
          "SEQ=%d|ID=%d|PAYLOAD=Données_test_%d|TIME=%lu",
          seqno, node_id, seqno, (unsigned long)clock_time());
        udp_packet_send(client_conn, buf, strlen(buf));
        printf("ENVOI: %s\n", buf);
        waiting_for_ack = 1;
        retransmissions = 0;
        etimer_set(&ack_timer, CLOCK_SECOND * 2);
      }
      else if(&ack_timer == data && waiting_for_ack) {
        retransmissions++;
        if(retransmissions < MAX_RETRANSMISSIONS) {
          printf("RETRANSMISSION #%d pour SEQ=%d\n",
                 retransmissions, seqno);
          udp_packet_send(client_conn, buf, strlen(buf));
          etimer_restart(&ack_timer);
        } else {
          printf("ÉCHEC: SEQ=%d perdu après %d tentatives\n",
                 seqno, MAX_RETRANSMISSIONS);
          waiting_for_ack = 0;
          etimer_restart(&periodic_timer);
        }
      }
    }
    else if(ev == tcpip_event && uip_newdata()) {
      if(strncmp((char*)uip_appdata, "ACK", 3) == 0) {
        int ack_seq;
        sscanf((char*)uip_appdata, "ACK:%d", &ack_seq);
        if(ack_seq == seqno && waiting_for_ack) {
          printf("ACK reçu pour SEQ=%d\n", seqno);
          waiting_for_ack = 0;
          etimer_set(&periodic_timer, SEND_INTERVAL);
        }
      }
    }
  }

  PROCESS_END();
}

4. Serveur UDP amélioré

Modifiez udp-server.c :

#include "contiki.h"
#include "net/ip/uip.h"
#include "net/ipv6/uip-ds6.h"
#include "net/ip/uip-udp-packet.h"
#include 
#include 

#define UDP_PORT 8765

PROCESS(udp_server_process, "UDP Server avancé");
AUTOSTART_PROCESSES(&udp_server_process);

static struct uip_udp_conn *server_conn;
static int total_packets = 0;
static int last_seq_per_node[10] = {0};

PROCESS_THREAD(udp_server_process, ev, data)
{
  uip_ipaddr_t client_ip;
  char response[32];
  int seq, id;
  PROCESS_BEGIN();

  server_conn = udp_new(NULL, UIP_HTONS(UDP_PORT), NULL);
  udp_bind(server_conn, UIP_HTONS(UDP_PORT));

  printf("Serveur UDP avancé démarré\n");

  while(1) {
    PROCESS_YIELD();
    if(ev == tcpip_event && uip_newdata()) {
      total_packets++;
      uip_ipaddr_copy(&client_ip, &UIP_IP_BUF->srcipaddr);

      printf("REÇU de ");
      uip_debug_ipaddr_print(&client_ip);
      printf(": %s\n", (char*)uip_appdata);

      // Extraire le numéro de séquence et l'ID
      if(sscanf((char*)uip_appdata, "SEQ=%d|ID=%d",
                &seq, &id) >= 2) {
        if(seq == last_seq_per_node[id] + 1) {
          printf("  → Paquet dans l'ordre (SEQ=%d)\n", seq);
        } else {
          printf("  → Paquet DÉSORDONNÉ (attendu %d, reçu %d)\n",
                 last_seq_per_node[id] + 1, seq);
        }
        last_seq_per_node[id] = seq;
      }

      // Envoyer un ACK
      snprintf(response, sizeof(response), "ACK:%d", seq);
      udp_packet_send(server_conn, response, strlen(response));
    }
  }

  PROCESS_END();
}

make TARGET=cooja clean

Créez une simulation avec 1 serveur + 2 clients. Démarrez et observez :

  • Les numéros de séquence s'incrémentent
  • Le serveur envoie des ACK
  • Les retransmissions en cas de perte

7. Exercices

  1. Ajoutez un champ "batterie simulée" dans le payload, qui décroît à chaque envoi
  2. Implémentez un mécanisme de fenêtre glissante (envoi de 3 paquets sans ACK)
  3. Ajoutez un détecteur de pertes : si 5 paquets consécutifs sont perdus, changez de parent RPL
