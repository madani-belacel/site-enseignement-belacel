3. Script d'analyse des logs de latence

Créez un fichier analyse_latence.py :

#!/usr/bin/env python3
import sys
import re
import matplotlib.pyplot as plt

def parse_logs(filename):
    tx_times = {}  # seqno -> time
    latencies = []

    with open(filename, 'r') as f:
        for line in f:
            parts = line.strip().split(';')
            if len(parts) < 4:
                continue
            if parts[0] == 'TX':
                # Format: TX;node_id;seqno;time
                seqno = int(parts[2])
                t = int(parts[3])
                tx_times[seqno] = t
            elif parts[0] == 'RX':
                # Format: RX;node_id;data;time
                data = parts[2]
                t_rx = int(parts[3])
                # Extraire seqno depuis le message "PING seqno time"
                match = re.search(r'PING (\d+)', data)
                if match:
                    seqno = int(match.group(1))
                    if seqno in tx_times:
                        lat = t_rx - tx_times[seqno]
                        latencies.append(lat)

    return latencies

def main():
    if len(sys.argv) < 2:
        print("Usage: analyse_latence.py ")
        return

    latencies = parse_logs(sys.argv[1])
    if not latencies:
        print("Aucune latence trouvée")
        return

    print(f"Nombre d'échantillons: {len(latencies)}")
    print(f"Latence moyenne: {sum(latencies)/len(latencies):.2f} ticks")
    print(f"Latence min: {min(latencies)} ticks")
    print(f"Latence max: {max(latencies)} ticks")

    plt.figure()
    plt.hist(latencies, bins=30, alpha=0.7)
    plt.xlabel("Latence (ticks)")
    plt.ylabel("Fréquence")
    plt.title("Distribution de la latence")
    plt.savefig("latence_histogram.png")
    print("Graphique sauvegardé: latence_histogram.png")

if __name__ == "__main__":
    main()

4. Script d'analyse des données PowerTracker

#!/usr/bin/env python3
import pandas as pd
import matplotlib.pyplot as plt

def analyse_energie(csv_file):
    df = pd.read_csv(csv_file)
    print("=== Résumé énergétique ===")
    print(df.groupby('Node')[['CPU(mJ)', 'RadioTX(mJ)', 'RadioRX(mJ)', 'Total(mJ)']].sum())

    # Graphique par nœud
    total_par_noeud = df.groupby('Node')['Total(mJ)'].sum()
    plt.figure()
    total_par_noeud.plot(kind='bar')
    plt.xlabel("Nœud")
    plt.ylabel("Énergie totale (mJ)")
    plt.title("Consommation énergétique par nœud")
    plt.tight_layout()
    plt.savefig("energie_par_noeud.png")

    # Répartition moyenne CPU/Radio
    moyens = df[['CPU(mJ)', 'RadioTX(mJ)', 'RadioRX(mJ)', 'LPM(mJ)']].mean()
    plt.figure()
    moyens.plot(kind='pie', autopct='%1.1f%%')
    plt.title("Répartition moyenne de l'énergie")
    plt.savefig("repartition_energie.png")
    print("Graphiques sauvegardés.")

if __name__ == "__main__":
    import sys
    if len(sys.argv) < 2:
        print("Usage: analyse_energie.py ")
    else:
        analyse_energie(sys.argv[1])

5. Script de calcul du PDR

#!/usr/bin/env python3
import sys
from collections import defaultdict

def calcul_pdr(logfile):
    tx_count = defaultdict(int)
    rx_count = defaultdict(int)

    with open(logfile, 'r') as f:
        for line in f:
            parts = line.strip().split(';')
            if len(parts) < 2:
                continue
            if parts[0] == 'TX':
                node = int(parts[1])
                tx_count[node] += 1
            elif parts[0] == 'RX':
                node = int(parts[1])
                rx_count[node] += 1

    print("=== PDR par nœud ===")
    total_tx = sum(tx_count.values())
    total_rx = sum(rx_count.values())
    for node in sorted(set(list(tx_count.keys()) + list(rx_count.keys()))):
        tx = tx_count.get(node, 0)
        rx = rx_count.get(node, 0)
        pdr = (rx / tx * 100) if tx > 0 else 0
        print(f"Nœud {node}: TX={tx}, RX={rx}, PDR={pdr:.1f}%")

    pdr_global = (total_rx / total_tx * 100) if total_tx > 0 else 0
    print(f"\nPDR global: {pdr_global:.1f}%")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: calcul_pdr.py ")
    else:
        calcul_pdr(sys.argv[1])

6. Script d'analyse comparative

Pour comparer plusieurs protocoles (standard RPL vs RPLCB par exemple) :

#!/usr/bin/env python3
import matplotlib.pyplot as plt

protocoles = ['Standard RPL', 'RPLCB', 'RPL-AER']
pdr_values = [92.5, 96.3, 98.1]
latence_values = [45.2, 38.7, 32.1]
energie_values = [12.8, 14.2, 11.5]

fig, axes = plt.subplots(1, 3, figsize=(12, 4))
axes[0].bar(protocoles, pdr_values, color=['blue', 'green', 'red'])
axes[0].set_ylabel('PDR (%)')
axes[0].set_title('Taux de livraison')

axes[1].bar(protocoles, latence_values, color=['blue', 'green', 'red'])
axes[1].set_ylabel('Latence (ms)')
axes[1].set_title('Latence moyenne')

axes[2].bar(protocoles, energie_values, color=['blue', 'green', 'red'])
axes[2].set_ylabel('Énergie (mJ)')
axes[2].set_title('Consommation énergétique')

plt.tight_layout()
plt.savefig('comparaison_protocoles.png')
print("Graphique comparatif sauvegardé.")

7. Automatisation de l'analyse

#!/bin/bash
# analyse_complete.sh
LOGS="logs_simulation.csv"

echo "=== Analyse complète ==="
python3 calcul_pdr.py $LOGS
python3 analyse_latence.py $LOGS

echo "=== Terminé ==="
