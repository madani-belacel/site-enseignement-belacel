4. Guided code — Flask (20 min)

# app.py
from flask import Flask, render_template, request, redirect
from modele import Eleve, Note
from database import init_db, ajouter_eleve, get_eleves

app = Flask(__name__)

@app.route("/")
def index():
    return render_template("index.html")

@app.route("/eleves")
def liste_eleves():
    eleves = get_eleves()
    return render_template("eleves.html", eleves=eleves)

@app.route("/ajouter", methods=["POST"])
def ajouter():
    nom = request.form["nom"]
    prenom = request.form["prenom"]
    groupe = request.form["groupe"]
    ajouter_eleve(nom, prenom, groupe)
    return redirect("/eleves")

if __name__ == "__main__":
    init_db()
    app.run(debug=True)

5. OOP Model (15 min)

# modele.py
class Eleve:
    def __init__(self, id, nom, prenom, groupe):
        self.id = id
        self.nom = nom
        self.prenom = prenom
        self.groupe = groupe
        self.notes = []

    def moyenne(self):
        if not self.notes:
            return 0.0
        return sum(n.valeur for n in self.notes) / len(self.notes)

    def ajouter_note(self, matiere, valeur):
        if 0 <= valeur <= 20:
            self.notes.append(Note(self.id, matiere, valeur))

class Note:
    def __init__(self, eleve_id, matiere, valeur):
        self.eleve_id = eleve_id
        self.matiere = matiere
        self.valeur = valeur

6. Database (10 min)

# database.py
import sqlite3

DB = "donnees/ecole.db"

def init_db():
    conn = sqlite3.connect(DB)
    c = conn.cursor()
    c.execute("""
        CREATE TABLE IF NOT EXISTS eleves (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nom TEXT NOT NULL,
            prenom TEXT NOT NULL,
            groupe TEXT
        )
    """)
    c.execute("""
        CREATE TABLE IF NOT EXISTS notes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            eleve_id INTEGER NOT NULL,
            matiere TEXT NOT NULL,
            valeur REAL NOT NULL CHECK(valeur >= 0 AND valeur <= 20),
            FOREIGN KEY(eleve_id) REFERENCES eleves(id)
        )
    """)
    conn.commit()
    conn.close()

def get_eleves():
    conn = sqlite3.connect(DB)
    conn.row_factory = sqlite3.Row
    c = conn.cursor()
    c.execute("SELECT * FROM eleves")
    eleves = [dict(row) for row in c.fetchall()]
    conn.close()
    return eleves

Pedagogical Note ENS

Teaching transposition : The project methodology presented here (specifications, architecture, sprints) can be transposed to middle/high school in the form of a Scratch or Python project. The important thing is to follow the same approach: 1) define the need, 2) design the solution, 3) implement by iterations, 4) test, 5) present. In NSI Terminale, the project is a baccalaureate exam — this session thus directly prepares future teachers to support their students.

Trainer advice : The ENS2 project is the moment of truth for the module. Ensure that each student has a functional minimal project before aiming for advanced features ("Make it work, then make it better"). Encourage collaboration but verify individual contributions via Git (each student must commit). Schedule a peer code review session before the defense.

Extension : Students can extend the project with advanced features: CSV import (to load an existing class), grade distribution chart generation, sending grade reports by email, or an administration interface with login. These extensions can be presented as "bonuses" during the defense and valued in the grade.
