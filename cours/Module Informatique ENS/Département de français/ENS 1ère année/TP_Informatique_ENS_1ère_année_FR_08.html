Exercice 2 — Module de chargement CSV (15 min)

Énoncé : Créer les fonctions de chargement des questions dans quiz.py : 1. charger_questions(fichier) — charge le CSV et retourne une liste de dictionnaires 2. filtrer_par_categorie(questions, categorie) — filtre les questions par catégorie 3. melanger_questions(questions) — mélange aléatoirement

Corrigé :

import csv
import random
import time
from datetime import date

def charger_questions(fichier):
    """Charge les questions depuis un fichier CSV."""
    questions = []
    try:
        with open(fichier, "r", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            for ligne in reader:
                questions.append({
                    "categorie": ligne["Categorie"],
                    "question": ligne["Question"],
                    "reponse": ligne["Reponse"]
                })
    except FileNotFoundError:
        print(f"Erreur : fichier {fichier} introuvable.")
        return []
    return questions

def filtrer_par_categorie(questions, categorie):
    """Filtre les questions par catégorie."""
    if categorie == "Toutes":
        return questions
    return [q for q in questions if q["categorie"] == categorie]

def melanger_questions(questions):
    """Mélange les questions aléatoirement."""
    random.shuffle(questions)
    return questions

def menu_principal():
    """Affiche le menu principal et gère les choix."""
    questions = charger_questions("questions.csv")
    if not questions:
        return
    categories = ["Toutes", "Architecture", "Algorithmique", "Python", "Réseaux"]
    while True:
        print("\n" + "="*40)
        print("1. Lancer le quiz")
        print("2. Règles")
        print("3. Quitter")
        choix = input("Votre choix : ")
        if choix == "1":
            for i, cat in enumerate(categories, 1):
                print(f"{i}. {cat}")
            try:
                c = int(input("Catégorie (1-5) : "))
                cat = categories[c - 1] if 1 <= c <= 5 else "Toutes"
            except (ValueError, IndexError):
                cat = "Toutes"
            qf = filtrer_par_categorie(questions, cat)
            print(f"Questions : {len(qf)} dans '{cat}'")
            lancer_quiz(qf, input("Votre nom : "))
        elif choix == "2":
            print("• Réponse textuelle • 4 catégories • Score /20 • Export CSV")
        elif choix == "3":
            print("Au revoir !")
            break
        else:
            print("Choix invalide.")

Exercice 4 — Tirage aléatoire et scoring (15 min)

Énoncé : Ajouter les fonctions de quiz avec tirage aléatoire, scoring et mention. Le programme mélange les questions, pose chaque question, vérifie la réponse, et attribue une mention.

Corrigé :

def poser_question(question):
    """Pose une question et retourne True si bonne réponse."""
    print(f"\n[{question['categorie']}] {question['question']}")
    reponse = input("Votre réponse : ").strip().lower()
    reponse_attendue = question['reponse'].strip().lower()
    if reponse == reponse_attendue:
        print("✓ Correct !")
        return True
    else:
        print(f"✗ Faux. Réponse : {question['reponse']}")
        return False

def lancer_quiz(questions, nom_eleve=""):
    """Lance le quiz avec mélange aléatoire et chronomètre."""
    if not questions:
        print("Aucune question à poser.")
        return

    questions = melanger_questions(questions)
    score = 0
    total = len(questions)
    debut = time.time()

    print(f"\n=== Quiz lancé ! ({total} questions) ===")
    for i, q in enumerate(questions, 1):
        print(f"\nQuestion {i}/{total}")
        if poser_question(q):
            score += 1

    temps = time.time() - debut
    afficher_resultats(score, total, temps)
    exporter = input("\nExporter les résultats ? (o/n) : ")
    if exporter == "o" and nom_eleve:
        exporter_resultats(score, total, temps, nom_eleve)

def afficher_resultats(score, total, temps):
    """Affiche les résultats avec mention."""
    pourcentage = (score / total * 100) if total > 0 else 0
    print("\n" + "="*40)
    print("           RÉSULTATS")
    print("="*40)
    note = round(score / total * 20, 2) if total > 0 else 0
    print(f"Score : {score}/{total} ({pourcentage:.1f}%)")
    print(f"Note : {note}/20")
    print(f"Temps : {temps:.1f}s")

    if pourcentage == 100:
        mention = "Parfait !"
    elif pourcentage >= 80:
        mention = "Très bien !"
    elif pourcentage >= 60:
        mention = "Bien !"
    elif pourcentage >= 40:
        mention = "Peut mieux faire."
    else:
        mention = "À revoir."
    print(f"Mention : {mention}")

def exporter_resultats(score, total, temps, nom_eleve, nom_fichier="resultats.csv"):
    """Exporte les résultats du quiz dans un fichier CSV."""
    with open(nom_fichier, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        if f.tell() == 0:
            writer.writerow(["Date", "Eleve", "Score", "Total",
                            "Pourcentage", "Note_20", "Temps_s"])
        note = round(score / total * 20, 2) if total > 0 else 0
        pourcentage = f"{score/total*100:.1f}" if total > 0 else "0"
        writer.writerow([date.today(), nom_eleve, score, total,
                        pourcentage, note, f"{temps:.1f}"])
    print(f"Résultats exportés dans {nom_fichier}")

Note pédagogique ENS

Transposition didactique

Ce mini-projet est l'aboutissement du module. Il mobilise toutes les compétences acquises : variables, conditions, boucles, fonctions, listes, dictionnaires, fichiers CSV, et gestion d'erreurs. - Pour les futurs enseignants : Ce quiz est un outil directement réutilisable. Ils peuvent adapter les questions à leur matière (histoire, langues, sciences) et utiliser l'export CSV pour suivre la progression des élèves. - En classe de secondaire : Simplifier en ne gardant qu'une seule catégorie de questions.

Conseil formateur

Accompagner les étudiants dans la structuration du projet : découpage en fonctions, fichier de données séparé, gestion des erreurs. Le projet peut être réalisé en binômes (programmation en paire). Évaluer sur la qualité du code (commentaires, nommage) autant que sur le fonctionnement. Pour les élèves de lycée, alléger le cahier des charges en supprimant l'export CSV.
